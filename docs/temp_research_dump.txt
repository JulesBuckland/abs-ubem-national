def pull_manchester_weather():

    # Manchester Coordinates
    lat, lon = 53.4808, -2.2426
    start_date = "2022-12-01"
    end_date = "2022-12-14" # 14 days (336 hours)
    
    url = f"https://archive-api.open-meteo.com/v1/archive?latitude={lat}&longitude={lon}&start_date={start_date}&end_date={end_date}&hourly=temperature_2m"

        response = requests.get(url)

def get_pm25_annual_mean(lat, lon):
    url = "https://air-quality-api.open-meteo.com/v1/air-quality"
    params = {
        "latitude": lat,
        "longitude": lon,
        "hourly": "pm2_5",
        "start_date": "2022-01-01",
        "end_date": "2022-12-31"
    }
    try:
        response = requests.get(url, params=params)
        response.raise_for_status()
        data = response.json()
        return np.mean(data['hourly']['pm2_5'])
    except Exception as e:
        logger.error(f"Failed to pull PM2.5 for {lat}, {lon}: {e}")
        return 8.5 # Regional average fallback

def update_confounders():
    logger.info("Updating MSOA Confounders with Real Overcrowding and PM2.5 data...")
    
    # 1. Process Overcrowding from TS052
    ts052_path = RAW_DIR / "ts052_msoa" / "census2021-ts052-msoa.csv"
    if not ts052_path.exists():
        raise FileNotFoundError(f"TS052 data not found at {ts052_path}")
    
    overcrowd_df = pd.read_csv(ts052_path)
    total_col = 'Occupancy rating for bedrooms: Total: All households'
    neg1_col = 'Occupancy rating for bedrooms: Occupancy rating of bedrooms: -1'
    neg2_col = 'Occupancy rating for bedrooms: Occupancy rating of bedrooms: -2 or less'
    
    overcrowd_df['overcrowding_prev'] = (overcrowd_df[neg1_col] + overcrowd_df[neg2_col]) / overcrowd_df[total_col]
    overcrowd_df = overcrowd_df[['geography code', 'overcrowding_prev']]
    overcrowd_df.columns = ['msoa_cd', 'overcrowding_prev']
    
    # 2. Pull PM2.5 for each LAD
    lad_pm25 = []
    for lad, coords in LAD_COORDS.items():
        logger.info(f"Pulling PM2.5 for {lad}...")
        mean_pm25 = get_pm25_annual_mean(coords[0], coords[1])
        lad_pm25.append({'ladnm': lad, 'pm25_val': mean_pm25})
    
    pm25_df = pd.DataFrame(lad_pm25)
    
    # 3. Map PM2.5 to MSOA
    lookup = pd.read_csv(RAW_DIR / "lookup.csv", usecols=['msoa21cd', 'ladnm']).drop_duplicates()
    msoa_pm25 = lookup.merge(pm25_df, on='ladnm')[['msoa21cd', 'pm25_val']].drop_duplicates()
    
    # 4. Merge all into existing confounders
    conf = pd.read_csv(PROCESSED_DIR / "msoa_confounders.csv")
    
    # Drop old overcrowding_prev and no2_val if they exist (we're replacing no2 with pm25)
    cols_to_drop = [c for c in ['overcrowding_prev', 'no2_val', 'pm25_val'] if c in conf.columns]
    conf = conf.drop(columns=cols_to_drop)
    
    conf = conf.merge(overcrowd_df, on='msoa_cd', how='left')
    conf = conf.merge(msoa_pm25, left_on='msoa_cd', right_on='msoa21cd', how='left').drop(columns=['msoa21cd'])
    
    # Fill missing values
    conf['overcrowding_prev'] = conf['overcrowding_prev'].fillna(conf['overcrowding_prev'].median())
    conf['pm25_val'] = conf['pm25_val'].fillna(conf['pm25_val'].median())
    
    conf.to_csv(PROCESSED_DIR / "msoa_confounders.csv", index=False)
    logger.info(f"Updated confounders saved. Head:\n{conf.head()}")

if __name__ == "__main__":
    update_confounders()

## scripts/02_bayesian_calibration.py

import pandas as pd
import numpy as np
import pymc as pm
import arviz as az
from config import PROCESSED_DIR, RANDOM_SEED
import logging

logging.basicConfig(level=logging.INFO, format='%(levelname)s: %(message)s')
logger = logging.getLogger("BayesianCalibration")

def run_calibration():
    logger.info("--- STAGE 2: JOINT HIERARCHICAL CALIBRATION ---")
    
    gas = pd.read_csv(PROCESSED_DIR / "gm_gas_2022.csv")
    mix = pd.read_csv(PROCESSED_DIR / "lsoa_archetype_mix.csv")
    imd = pd.read_csv(PROCESSED_DIR / "lsoa_imd_gm.csv")
    priors_raw = pd.read_csv(PROCESSED_DIR / "gm_archetype_priors.csv").set_index('Archetype')
    census = pd.read_csv(PROCESSED_DIR / "gm_building_stock_census.csv")

    # Align spatial datasets
    common = sorted(list(set(gas['lsoa_cd']) & set(mix['LSOA21CD']) & set(imd['lsoa_cd']) & set(census['lsoa_cd'])))
    gas = gas[gas['lsoa_cd'].isin(common)].sort_values('lsoa_cd')
    mix = mix[mix['LSOA21CD'].isin(common)].sort_values('LSOA21CD')
    imd = imd[imd['lsoa_cd'].isin(common)].sort_values('lsoa_cd')
    census = census[census['lsoa_cd'].isin(common)].sort_values('lsoa_cd')

    available_archetypes = [a for a in mix.columns if a in priors_raw.index]
    
    # Model Inputs
    X = (census.iloc[:, 1].values.reshape(-1, 1) * 
         (mix[available_archetypes].values / 100.0) * 
         priors_raw.loc[available_archetypes, 'Mean_Area'].values.reshape(1, -1)) / 1e6
    y_obs = gas['annual_gas_kwh'].values / 1e6
    income_scores = imd['income_score'].values

    # Priors
    prior_means = np.array([200.0] * len(available_archetypes)) # Generalized fallback

    with pm.Model() as model:
        # Building Intensities
        intensities = pm.TruncatedNormal("intensities", mu=prior_means, sigma=50.0, lower=20.0, shape=len(available_archetypes))
        # Behavioral Rationing (Global)
        beta_dep = pm.Normal("beta_dep", mu=1.0, sigma=0.5)
        # Sigmoid Multiplier [0.5, 1.0]
        b_mult = 0.5 + 0.5 * pm.math.sigmoid(-beta_dep * income_scores)
        
        # Likelihood
        mu = pm.math.dot(X, intensities) * b_mult
        sigma = pm.HalfNormal("sigma", sigma=1.0)
        pm.Normal("y", mu=mu, sigma=sigma, observed=y_obs)
        
        trace = pm.sample(1000, tune=500, target_accept=0.95, chains=4, random_seed=RANDOM_SEED)
        az.to_netcdf(trace, str(PROCESSED_DIR / "calibration_trace.nc"))
        
    logger.info("Calibration Complete. Trace saved.")

if __name__ == "__main__":
    run_calibration()


## scripts/03_thermodynamic_simulation.py

import pandas as pd
import numpy as np
import arviz as az
import statsmodels.api as sm
from scipy.special import expit
from tqdm import tqdm
from config import RAW_DIR, PROCESSED_DIR, RANDOM_SEED, N_HH_PER_MSOA, N_REALIZATIONS, get_cold_snap_temps
import logging

logging.basicConfig(level=logging.INFO, format='%(levelname)s: %(message)s')
logger = logging.getLogger("ThermoSimulation")

def run_simulation():
    logger.info("--- STAGE 3: VECTORIZED THERMODYNAMICS & HEALTH ANALYSIS ---")
    np.random.seed(RANDOM_SEED)
    
    # 1. Load Bayesian Trace & Priors
    trace = az.from_netcdf(str(PROCESSED_DIR / "calibration_trace.nc"))
    post_intensities = trace.posterior['intensities'].values.reshape(-1, trace.posterior['intensities'].shape[-1])
    post_beta_dep = trace.posterior['beta_dep'].values.flatten()
    n_samples = post_intensities.shape[0]
    
    indices_df = pd.read_csv(PROCESSED_DIR / "archetype_indices.csv").set_index('archetype')
    priors = pd.read_csv(PROCESSED_DIR / "gm_archetype_priors.csv").set_index('Archetype')
    
    # 2. Load Spatial & Demographic Data
    mix = pd.read_csv(PROCESSED_DIR / "lsoa_archetype_mix.csv")
    lookup = pd.read_csv(RAW_DIR / "lookup.csv", usecols=['lsoa21cd', 'msoa21cd', 'ladnm']).drop_duplicates()
    
    ehs = pd.read_csv(RAW_DIR / "ehs_2020_household_fuel_poverty.tab", sep='\t')
    microdata = ehs[['hhincx', 'tenex', 'hhsizex']].dropna().copy()
    microdata.columns = ['income', 'tenure', 'hh_size']
    
    temps = get_cold_snap_temps()
    
    # 3. Vectorized Simulation Loop
    results = []
    msoas = mix.merge(lookup, left_on='LSOA21CD', right_on='lsoa21cd')['msoa21cd'].unique()
    msoa_archetypes = mix.merge(lookup, left_on='LSOA21CD', right_on='lsoa21cd').groupby('msoa21cd').mean(numeric_only=True)
    archetype_list = msoa_archetypes.columns.tolist()
    
    # Load confounders for socio-economic injection (Scientific Proof Fix)
    conf = pd.read_csv(PROCESSED_DIR / "msoa_confounders.csv").set_index('msoa_cd')

    for msoa in tqdm(msoas, desc="Simulating MSOAs"):
        # Sampling Logic
        arch_weights = msoa_archetypes.loc[msoa].values
        arch_idx = np.random.choice(len(archetype_list), size=(N_REALIZATIONS, N_HH_PER_MSOA), p=arch_weights/arch_weights.sum())
        arch_names = np.array(archetype_list)[arch_idx]
        
        # Paired Posterior Sampling (Hazard 1 Fix)
        trace_idx = np.random.choice(n_samples, size=(N_REALIZATIONS, N_HH_PER_MSOA))
        
        flat_arch_names = arch_names.flatten()
        flat_trace_idx = trace_idx.flatten()
        
        # Pull Alphas and Betas
        arch_trace_indices = indices_df.loc[flat_arch_names, 'intensity_idx'].values
        alphas = post_intensities[flat_trace_idx, arch_trace_indices].reshape(N_REALIZATIONS, N_HH_PER_MSOA)
        beta_samples = post_beta_dep[flat_trace_idx].reshape(N_REALIZATIONS, N_HH_PER_MSOA)
        
        # Socio-Economic Attributes (Baking in the Income Penalty)
        msoa_income_dep = conf.loc[msoa, 'income_dep_score'] if msoa in conf.index else 0.2
        scaled_income = np.random.normal(1.0 - msoa_income_dep, 0.1, size=(N_REALIZATIONS, N_HH_PER_MSOA))
        scaled_income = np.clip(scaled_income, 0.01, 0.99)
        
        # Sigmoid Bounding (Behavioral Rationing Multiplier)
        b_mults = 0.5 + 0.5 * expit(-beta_samples * (1.0 - scaled_income))
        
        # Thermodynamic Physics
        areas = np.maximum(np.random.normal(priors.loc[flat_arch_names, 'Mean_Area'].values.reshape(N_REALIZATIONS, N_HH_PER_MSOA), 10.0), 15.0)
        hlcs = (alphas * areas) * 1000 / (2500 * 24)
        q_ints = (2.0 * areas) + (35.0 * 2.4) + 40.0 
        
        # WFH Calculation (Vectorized over Time)
        t_out = temps[np.newaxis, np.newaxis, :]
        t_no_heat = t_out + (q_ints[:, :, np.newaxis] / hlcs[:, :, np.newaxis])
        
        # Physics-Only WFH (b_mults = 1.0, assuming standard 18°C setpoint achieved)
        q_heat_phys = np.minimum(6000.0, np.maximum(0, (18.0 - t_no_heat) * hlcs[:, :, np.newaxis]))
        t_ins_phys = t_out + (q_heat_phys + q_ints[:, :, np.newaxis]) / hlcs[:, :, np.newaxis]
        wfh_phys = np.where(t_ins_phys < 18.0, 18.0 - t_ins_phys, 0.0).sum(axis=2).mean()

        # Integrated WFH (Applying income-rationing behavioral constraints)
        q_heat_int = q_heat_phys * b_mults[:, :, np.newaxis]
        t_ins_int = t_out + (q_heat_int + q_ints[:, :, np.newaxis]) / hlcs[:, :, np.newaxis]
        wfh_int = np.where(t_ins_int < 18.0, 18.0 - t_ins_int, 0.0).sum(axis=2).mean()
        
        results.append({
            'msoa_cd': msoa, 
            'wfh_dh_mean': wfh_int, 
            'wfh_physical_only': wfh_phys, 
            'hlc_mean': hlcs.mean()
        })

    res_df = pd.DataFrame(results)
    res_df.to_csv(PROCESSED_DIR / "simulated_wfh_results.csv", index=False)

    # 4. Final Comparative Regression Face-Off
    logger.info("Running Comparative Regression Face-Off with Spatial Diagnostics (V8)...")
    copd = pd.read_csv(RAW_DIR / "gm_copd_admissions_unified.csv")
    copd.columns = ['msoa_cd', 'copd_sar']
    conf = pd.read_csv(PROCESSED_DIR / "msoa_confounders.csv")
    
    data = res_df.merge(copd, on='msoa_cd').merge(conf, on='msoa_cd').dropna()
    y = data['copd_sar']

    def standardize(series):
        return (series - series.mean()) / series.std()

    # Z-score standardization
    data['smoking_z'] = standardize(data['smoking_prev'])
    data['wfh_phys_z'] = standardize(data['wfh_physical_only'])
    data['wfh_int_z'] = standardize(data['wfh_dh_mean'])
    data['income_z'] = standardize(data['income_dep_score'])
    data['age_z'] = standardize(data['pct_over_65'])
    data['pm25_z'] = standardize(data['pm25_val'])
    data['overcrowd_z'] = standardize(data['overcrowding_prev'])

    # --- SPATIAL PREP ---
    import libpysal
    import geopandas as gpd
    msoa_bounds = gpd.read_file(RAW_DIR / "spatial" / "MSOA_2021_EW_BGC_V3.shp")
    data_msoas = data['msoa_cd'].unique()
    msoa_bounds = msoa_bounds[msoa_bounds['MSOA21CD'].isin(data_msoas)].sort_values('MSOA21CD')
    data = data.sort_values('msoa_cd')
    
    w = libpysal.weights.Queen.from_dataframe(msoa_bounds)
    w.transform = 'r'

    def get_moran(resid, w):
        z = (resid - resid.mean())
        return (z * w.sparse.dot(z)).sum() / (z**2).sum()

    # Model A: Standard Deterministic Physics
    X_a = sm.add_constant(data[['smoking_z', 'wfh_phys_z']])
    model_a = sm.OLS(data['copd_sar'], X_a).fit()
    moran_a = get_moran(model_a.resid, w)

    # Model B: Integrated Digital Twin (Base Theory)
    X_b = sm.add_constant(data[['smoking_z', 'wfh_int_z']])
    model_b = sm.OLS(data['copd_sar'], X_b).fit()
    moran_b = get_moran(model_b.resid, w)

    # Model C: Social Only (Income + Smoking)
    X_c = sm.add_constant(data[['smoking_z', 'income_z']])
    model_c = sm.OLS(data['copd_sar'], X_c).fit()
    moran_c = get_moran(model_c.resid, w)

    # Model D: FINAL OPTIMIZED DIGITAL TWIN (Parsimonious V8)
    X_d_cols = ['smoking_z', 'wfh_int_z', 'age_z', 'pm25_z', 'overcrowd_z']
    X_d = sm.add_constant(data[X_d_cols])
    model_d = sm.OLS(data['copd_sar'], X_d).fit()
    moran_d = get_moran(model_d.resid, w)

    # --- GEOGRAPHY HACK: SPATIAL ERROR MODEL (SEM) ---
    logger.info("Implementing Spatial Error Model (SEM) for Model D refinement...")
    try:
        from spreg import ML_Error
        y_sem = data['copd_sar'].values.reshape(-1, 1)
        X_sem = data[X_d_cols].values
        model_sem = ML_Error(y_sem, X_sem, w=w, name_y='copd_sar', name_x=X_d_cols)
        sem_r2 = model_sem.pr2
        sem_aic = model_sem.aic
        # Moran's I on SEM residuals (filtered to check for remaining autocorrelation)
        u_sem = model_sem.u
        lambda_sem = model_sem.lam
        epsilon = u_sem - lambda_sem * (w.sparse.dot(u_sem))
        moran_e = get_moran(epsilon.flatten(), w)
    except Exception as e:
        logger.error(f"Spatial modeling failed: {e}")
        sem_r2, sem_aic, moran_e = 0, 0, 0

    # --- MULTICOLLINEARITY CHECK (VIF) ---
    from statsmodels.stats.outliers_influence import variance_inflation_factor
    X_vif = data[X_d_cols]
    vif_data = pd.DataFrame()
    vif_data["feature"] = X_vif.columns
    vif_data["VIF"] = [variance_inflation_factor(sm.add_constant(X_vif).values, i+1) for i in range(len(X_vif.columns))]
    
    logger.info("Variance Inflation Factors (VIF):")
    for _, row in vif_data.iterrows():
        logger.info(f"  {row['feature']}: {row['VIF']:.2f}")

    # Log Comparison Results
    logger.info(f"Model A: R2={model_a.rsquared:.3f}, Moran's I={moran_a:.3f}")
    logger.info(f"Model B: R2={model_b.rsquared:.3f}, Moran's I={moran_b:.3f}")
    logger.info(f"Model C: R2={model_c.rsquared:.3f}, Moran's I={moran_c:.3f}")
    logger.info(f"Model D: R2={model_d.rsquared:.3f}, Moran's I={moran_d:.3f}")
    if sem_r2 > 0:
        logger.info(f"Model E: R2={sem_r2:.3f}, Moran's I={moran_e:.3f} (FIXED)")
    
    # Save the full report for paper results
    with open(PROCESSED_DIR / "final_regression_report.txt", "w") as f:

## Abstract
Traditional deterministic models of domestic energy use often fail to capture the "Performance Gap" between theoretical building physics and real-world occupant behavior, particularly for low-income households. This study introduces the **Thermal Vulnerability Index (TVI)**—a probabilistic metric derived from "Weighted Failure Hours" (WFH)—to quantify the chronic health risks embedded in the domestic building stock. Using hierarchical Bayesian calibration and individual-level stock estimation, we recover physical housing signals from empirical DESNZ gas consumption and a representative cold-snap stress test (December 2022). Our Spatial Error Model (SEM) demonstrates that the TVI is the primary structural determinant of respiratory health outcomes (Coeff = 69.23, p < 0.001, Pseudo R-squared = 0.688), far outweighing air quality. We validate the physical model against empirical 2022 gas data ($r = -0.312$), confirming the activation of a "Age-Built-Form Paradox" where thermodynamic inefficiency meets economic constraint. These results provide evidence that health equity requires lowering the thermodynamic barrier of survival through targeted, high-performance retrofit.

## 1. Introduction
The thermal performance of the domestic building stock is a critical, yet often overlooked, mediator of socioeconomic health inequalities. Existing research often relies on deterministic MSOA-level averages or EPC ratings, which mask significant heterogeneities in both physical fabric and occupant behavior. This study addresses these limitations by adopting a probabilistic "Digital Twin" approach, following the methodological rigor established by the Cambridge Energy Efficient Cities Initiative (Booth et al., 2012; Neto-Bradley et al., 2022).

We hypothesize the existence of a **"Fabric-Vulnerability Paradox"**: health outcomes are not primarily driven by the poor living in lower-quality housing ("Architectural Sorting"), but rather by the interaction of a physically uniform, inefficient building stock with varying economic capacities to overcome that inefficiency. This creates a **"Age-Built-Form Paradox"**, where the thermodynamic threshold for health is socially determined by income.

## 1. Introduction
The thermal performance of the domestic building stock is a critical, yet often overlooked, mediator of socioeconomic health inequalities. Existing research often relies on deterministic MSOA-level averages or EPC ratings, which mask significant heterogeneities in both physical fabric and occupant behavior. This study addresses these limitations by adopting a probabilistic "Digital Twin" approach, following the methodological rigor established by the Cambridge Energy Efficient Cities Initiative (Booth et al., 2012; Neto-Bradley et al., 2022).

We hypothesize the existence of a **"Fabric-Vulnerability Paradox"**: health outcomes are not primarily driven by the poor living in lower-quality housing ("Architectural Sorting"), but rather by the interaction of a physically uniform, inefficient building stock with varying economic capacities to overcome that inefficiency. This creates a **"Age-Built-Form Paradox"**, where the thermodynamic threshold for health is socially determined by income.
